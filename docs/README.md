### Automation Script Example

```python
# Complete PR labeling automation
import requests
import time
import logging

class GitHubAutomation:
    def __init__(self, token, owner, repo):
        self.token = token
        self.owner = owner
        self.repo = repo
        self.headers = {
            'Authorization': f'token {token}',
            'Accept': 'application/vnd.github.v3+json'
        }
    
    def get_open_prs(self):
        """Fetch all open pull requests."""
        prs = []
        page = 1
        
        while True:
            url = f'https://api.github.com/repos/{self.owner}/{self.repo}/pulls'
            params = {'page': page, 'per_page': 100, 'state': 'open'}
            
            response = requests.get(url, headers=self.headers, params=params)
            response.raise_for_status()
            
            page_prs = response.json()
            if not page_prs:
                break
                
            prs.extend(page_prs)
            page += 1
        
        return prs
    
    def add_label(self, pr_number, label):
        """Add label to specific PR."""
        url = f'https://api.github.com/repos/{self.owner}/{self.repo}/issues/{pr_number}/labels'
        data = {'labels': [label]}
        
        response = requests.post(url, headers=self.headers, json=data)
        return response.status_code == 200
    
    def auto_label_prs(self, target_label='run dio workflow'):
        """Automatically label PRs that don't have the target label."""
        prs = self.get_open_prs()
        labeled_count = 0
        
        for pr in prs:
            existing_labels = [label['name'] for label in pr['labels']]
            
            if target_label not in existing_labels:
                if self.add_label(pr['number'], target_label):
                    labeled_count += 1
                    logging.info(f"Added label to PR #{pr['number']}")
                time.sleep(1)  # Rate limiting
        
        logging.info(f"Processed {len(prs)} PRs, labeled {labeled_count}")
        return labeled_count

# Usage
automation = GitHubAutomation('your-token', 'digitalinnovationone', 'dio-lab-open-source')
automation.auto_label_prs()
```
